Primer plottanja replikatov

rmsf_plot <- rmsf |>
    mutate(residue = seq_len(n())) |>
    tidyr::pivot_longer(cols = starts_with("RMSF_R"), names_to = "replicate", values_to = "rmsf")

ggplot(rmsf_plot, aes(x = residue, y = rmsf, color = replicate)) +
    geom_line(size = 0.5, alpha = 0.7) +
    geom_point(size = 1, alpha = 0.7) +
    scale_color_manual(
        values = c(RMSF_R1 = "#ff8e32", RMSF_R2 = "#cd68bb", RMSF_R3 = "#51c3cc"),
        labels = c(RMSF_R1 = "1", RMSF_R2 = "2", RMSF_R3 = "3")
    ) +
    labs(x = "Residue index", y = "RMSF") +
    theme_light(base_size = 18) +
    theme(
        panel.grid.minor = element_blank(),
        panel.grid.major = element_line(color = "#dfe2e5"),
        legend.position = "top",
        legend.title = element_text(face = "bold"),
        plot.title = element_text(face = "bold", hjust = 0.5)
    )
